        // 初始化世界
        function initWorld() {
            const worldMap = document.getElementById('worldMap');
            const cellSize = 100;
            const cols = 30;
            const rows = 20;
            
            // 使用柏林噪聲生成更自然的地形
            const generateTerrain = (x, y) => {
                const scale = 0.1;
                const noise1 = Math.sin(x * scale) * Math.cos(y * scale);
                const noise2 = Math.sin(x * scale * 2) * Math.cos(y * scale * 2) * 0.5;
                const value = (noise1 + noise2 + Math.random() * 0.3) / 1.8;
                
                // 海岸線
                if (y < 3 || y > 17 || x < 2 || x > 27) {
                    return value > -0.2 ? TERRAIN_TYPES.WATER : TERRAIN_TYPES.SAND;
                }
                // 山脈
                if (Math.abs(x - 15) < 3 && Math.abs(y - 10) < 4 && value > 0.3) {
                    return TERRAIN_TYPES.MOUNTAIN;
                }
                // 森林
                if (value > 0.4) {
                    return TERRAIN_TYPES.FOREST;
                }
                // 沙地
                if (value < -0.3) {
                    return TERRAIN_TYPES.SAND;
                }
                // 草地
                return TERRAIN_TYPES.GRASS;
            };
            
            // 生成地形
            for (let y = 0; y < rows; y++) {
                gameState.terrainMap[y] = [];
                for (let x = 0; x < cols; x++) {
                    const cell = document.createElement('div');
                    const terrainType = generateTerrain(x, y);
                    
                    cell.className = `absolute rounded-lg ${terrainType.class}`;
                    cell.style.width = `${cellSize}px`;
                    cell.style.height = `${cellSize}px`;
                    cell.style.left = `${x * cellSize}px`;
                    cell.style.top = `${y * cellSize}px`;
                    cell.style.opacity = gameState.dayNight === 'day' ? '0.4' : '0.2';
                    cell.dataset.terrain = terrainType.name;
                    cell.dataset.x = x;
                    cell.dataset.y = y;
                    
                    worldMap.appendChild(cell);
                    gameState.terrainMap[y][x] = terrainType;
                    
                    // 生成資源
                    if (terrainType.fertility > 0 && Math.random() < terrainType.fertility * 0.3) {
                        generateResource(x * cellSize + 50, y * cellSize + 50, terrainType);
                    }
                }
            }
            
            addEventLog('🌍 世界已創建，資源已散佈', 'system');
        }

        // 資源類
        class Resource {
            constructor(x, y, type, symbol) {
                this.id = Date.now() + Math.random();
                this.x = x;
                this.y = y;
                this.type = type;
                this.symbol = symbol;
                this.name = null; // 待 AI 發現並命名
                this.discovered = false;
                this.poisonous = Math.random() < type.danger;
                this.element = null;
                this.nutrition = type.nutrition;
                this.hydration = type.hydration;
            }
            
            createElement() {
                const resource = document.createElement('div');
                resource.className = 'resource-item absolute';
                resource.id = `resource-${this.id}`;
                resource.style.left = `${this.x}px`;
                resource.style.top = `${this.y}px`;
                resource.style.zIndex = '15';
                
                resource.innerHTML = `
                    <div class="text-2xl opacity-70 hover:opacity-100 transition">
                        ${this.symbol}
                    </div>
                `;
                
                this.element = resource;
                document.getElementById('worldMap').appendChild(resource);
            }
            
            consume(agent) {
                if (this.poisonous) {
                    agent.health -= 30;
                    agent.mood = 'sick';
                    agent.memory.push({
                        type: 'danger',
                        resource: this.symbol,
                        name: this.name,
                        location: { x: this.x, y: this.y }
                    });
                    addEventLog(`⚠️ ${agent.name} 吃了有毒的${this.name || '未知物'}！健康 -30`, 'survival');
                } else {
                    agent.hunger = Math.min(100, agent.hunger + this.nutrition);
                    agent.thirst = Math.min(100, agent.thirst + this.hydration);
                    agent.mood = 'happy';
                    agent.memory.push({
                        type: 'food',
                        resource: this.symbol,
                        name: this.name,
                        nutrition: this.nutrition,
                        location: { x: this.x, y: this.y }
                    });
                    addEventLog(`✅ ${agent.name} 吃了${this.name || this.symbol}，飢餓 +${this.nutrition}`, 'survival');
                }
                
                // 資源被消耗後消失
                if (this.element) {
                    this.element.remove();
                }
                const index = gameState.resources.indexOf(this);
                if (index > -1) {
                    gameState.resources.splice(index, 1);
                }
                
                // 一段時間後重新生成
                setTimeout(() => {
                    const terrain = gameState.terrainMap[Math.floor(this.y/100)]?.[Math.floor(this.x/100)];
                    if (terrain && terrain.fertility > 0) {
                        generateResource(this.x, this.y, terrain);
                    }
                }, 30000 / gameState.speed);
            }
        }

        // 生成資源
        function generateResource(x, y, terrain) {
            let type, symbol;
            
            // 根據地形選擇資源類型
            if (terrain === TERRAIN_TYPES.WATER) {
                type = RESOURCE_TYPES.FISH;
            } else if (terrain === TERRAIN_TYPES.FOREST) {
                const types = [RESOURCE_TYPES.FRUIT, RESOURCE_TYPES.MUSHROOM, RESOURCE_TYPES.ANIMAL];
                type = types[Math.floor(Math.random() * types.length)];
            } else if (terrain === TERRAIN_TYPES.GRASS) {
                const types = [RESOURCE_TYPES.PLANT, RESOURCE_TYPES.FRUIT, RESOURCE_TYPES.ANIMAL];
                type = types[Math.floor(Math.random() * types.length)];
            } else if (terrain === TERRAIN_TYPES.MOUNTAIN) {
                type = RESOURCE_TYPES.WATER_SOURCE;
            } else {
                return; // 沙地不生成資源
            }
            
            symbol = type.symbols[Math.floor(Math.random() * type.symbols.length)];
            
            // 添加一些位置偏移
            x += (Math.random() - 0.5) * 80;
            y += (Math.random() - 0.5) * 80;
            
            const resource = new Resource(x, y, type, symbol);
            resource.createElement();
            gameState.resources.push(resource);
        }

        // Langfuse 追踪
        async function trackWithLangfuse(agent, action, prompt, response, cost) {
            if (!gameState.langfuseKey || !gameState.langfuseUrl) return;
            
            try {
                const traceId = `trace-${agent.id}-${Date.now()}`;
                const generationId = `gen-${Date.now()}`;
                
                // Langfuse Ingestion API 格式
                const events = [
                    {
                        id: traceId,
                        type: "trace-create",
                        timestamp: new Date().toISOString(),
                        body: {
                            id: traceId,
                            name: `AI Decision - ${agent.name}`,
                            userId: agent.id,
                            metadata: {
                                agentName: agent.name,
                                model: agent.model,
                                action: action,
                                gameDay: gameState.currentDay,
                                gameHour: gameState.currentHour
                            },
                            tags: [action, agent.mood, ALL_MODELS.find(m => m.id === agent.model)?.vendor || 'Unknown']
                        }
                    },
                    {
                        id: generationId,
                        type: "generation-create",
                        timestamp: new Date().toISOString(),
                        body: {
                            id: generationId,
                            traceId: traceId,
                            name: action,
                            model: agent.model,
                            modelParameters: {
                                temperature: 0.8,
                                maxTokens: 150
                            },
                            input: prompt,
                            output: response,
                            usage: {
                                promptTokens: Math.ceil(prompt.length / 4),
                                completionTokens: Math.ceil(response.length / 4),
                                totalTokens: Math.ceil((prompt.length + response.length) / 4)
                            },
                            metadata: {
                                health: agent.health,
                                hunger: agent.hunger,
                                thirst: agent.thirst,
                                energy: agent.energy,
                                mood: agent.mood,
                                estimatedCost: cost
                            }
                        }
                    }
                ];

                // 發送到 Langfuse API
                const response_api = await fetch('https://cloud.langfuse.com/api/public/ingestion', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'Authorization': `Basic ${btoa(gameState.langfuseUrl + ':' + gameState.langfuseKey)}`
                    },
                    body: JSON.stringify({
                        batch: events,
                        metadata: {
                            sdk_name: "ai-ecosystem-simulator",
                            sdk_version: "1.0.0"
                        }
                    })
                });

                if (response_api.ok) {
                    console.log('✅ Langfuse 追踪成功:', traceId);
                } else {
                    const errorText = await response_api.text();
                    console.error('❌ Langfuse API 錯誤:', response_api.status, errorText);
                }
                
                // 同時存儲在本地以便查看
                if (!agent.langfuseTraces) agent.langfuseTraces = [];
                agent.langfuseTraces.push({
                    traceId: traceId,
                    timestamp: new Date().toISOString(),
                    action: action,
                    prompt: prompt,
                    response: response,
                    cost: cost
                });
                
            } catch (error) {
                console.error('Langfuse tracking failed:', error);
            }
        }

            // 呼叫 Stima API
            async callStimaAPI(prompt) {
                if (!gameState.apiKey) {
                    throw new Error('API Key 未設置');
                }
                
                try {
                    gameState.apiCallCount++;
                    const estimatedCost = (prompt.length / 4 + 150) * this.modelCost / 1000; // 粗略估算
                    gameState.totalCost += estimatedCost;
                    updateStats();
                    
                    const response = await fetch(`${STIMA_CONFIG.url}/chat/completions`, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                            'Authorization': `Bearer ${gameState.apiKey}`
                        },
                        body: JSON.stringify({
                            model: this.model,
                            messages: [
                                {
                                    role: 'system',
                                    content: `你是${this.name}，一個在虛擬生態系統中的AI生命體。${this.personality}`
                                },
                                {
                                    role: 'user',
                                    content: prompt
                                }
                            ],
                            max_tokens: 150,
                            temperature: 0.8
                        })
                    });
                    
                    if (!response.ok) {
                        throw new Error(`API request failed: ${response.status}`);
                    }
                    
                    const data = await response.json();
                    const llmResponse = data.choices[0].message.content;
                    
                    // 記錄到 Langfuse
                    await trackWithLangfuse(this, 'decision', prompt, llmResponse, estimatedCost);
                    
                    return this.parseAPIResponse(llmResponse);
                    
                } catch (error) {
                    console.error('Stima API 調用失敗:', error);
                    return this.fallbackThink();
                }
            }
            
            parseAPIResponse(response) {
                // 解析 LLM 回應
                const actions = ['explore', 'eat', 'drink', 'sleep', 'socialize', 'rest'];
                let selectedAction = 'explore';
                let thought = response.substring(0, 100);
                
                // 簡單的關鍵字匹配來決定行動
                if (response.includes('餓') || response.includes('食') || response.includes('eat')) {
                    selectedAction = 'eat';
                } else if (response.includes('渴') || response.includes('水') || response.includes('drink')) {
                    selectedAction = 'drink';
                } else if (response.includes('累') || response.includes('睡') || response.includes('sleep')) {
                    selectedAction = 'sleep';
                } else if (response.includes('朋友') || response.includes('交流') || response.includes('social')) {
                    selectedAction = 'socialize';
                } else if (response.includes('休息') || response.includes('rest')) {
                    selectedAction = 'rest';
                }
                
                return { action: selectedAction, thought };
            }
            
            fallbackThink() {
                // 基於生存需求的決策
                let action = 'explore';
                let thought = '系統思考...';
                
                // 優先級決策
                if (this.health < 30) {
                    thought = '我感覺很不舒服，需要治療';
                    action = 'rest';
                } else if (this.hunger < 30) {
                    thought = '好餓啊，需要找食物';
                    action = 'eat';
                } else if (this.thirst < 30) {
                    thought = '好渴，需要找水源';
                    action = 'drink';
                } else if (this.energy < 30 || (gameState.dayNight === 'night' && this.energy < 60)) {
                    thought = '好累，需要休息了';
                    action = 'sleep';
                } else if (Math.random() < 0.2 && this.findNearbyAgent()) {
                    thought = '想和其他人交流一下';
                    action = 'socialize';
                } else {
                    const thoughts = [
                        '這個世界真有趣',
                        '我要去探索新的地方',
                        '不知道前面有什麼',
                        '今天天氣不錯'
                    ];
                    thought = thoughts[Math.floor(Math.random() * thoughts.length)];
                    action = 'explore';
                }
                
                return { action, thought };
            }
            
            async think() {
                const perception = this.perceive();
                const prompt = this.buildPrompt(perception);
                
                try {
                    return await this.callStimaAPI(prompt);
                } catch (error) {
                    return this.fallbackThink();
                }
            }
            
            async update() {
                if (!this.isAlive) return;
                
                // 更新生存狀態
                this.updateSurvivalStats();
                
                // 檢查是否死亡
                if (this.health <= 0 || this.hunger <= 0 || this.thirst <= 0) {
                    this.die();
                    return;
                }
                
                // 睡眠中跳過行動
                if (this.isSleeping) {
                    this.sleep();
                    return;
                }
                
                // 思考下一步行動
                const decision = await this.think();
                
                // 記錄思考過程
                this.addThinkingLog(decision.action, 'N/A', decision.thought);
                
                // 執行行動
                switch(decision.action) {
                    case 'explore':
                        this.explore();
                        break;
                    case 'eat':
                        this.eat();
                        break;
                    case 'drink':
                        this.drink();
                        break;
                    case 'sleep':
                        this.startSleep();
                        break;
                    case 'socialize':
                        this.socialize();
                        break;
                    case 'rest':
                        this.rest();
                        break;
                    default:
                        this.wander();
                }
                
                // 顯示思考內容
                if (decision.thought && Math.random() > 0.6) {
                    this.speak(decision.thought);
                }
                
                // 更新顯示
                this.updateDisplay();
            }
            
            updateSurvivalStats() {
                // 根據時間流逝消耗生存值 (1分鐘 = 2小時)
                const timeRate = (2 / 60) * gameState.speed; // 每小時消耗率 * 2小時
                const consumptionRate = this.isSleeping ? 0.5 : timeRate;
                
                this.hunger = Math.max(0, this.hunger - 0.2 * consumptionRate);
                this.thirst = Math.max(0, this.thirst - 0.3 * consumptionRate);
                this.energy = this.isSleeping ? 
                    Math.min(100, this.energy + 2 * consumptionRate) : 
                    Math.max(0, this.energy - 0.15 * consumptionRate);
                
                // 飢餓和口渴影響健康
                if (this.hunger < 10) this.health -= 0.5;
                if (this.thirst < 10) this.health -= 0.8;
                if (this.energy < 10) this.health -= 0.3;
                
                // 更新心情
                if (this.health < 30) this.mood = 'sick';
                else if (this.hunger < 30) this.mood = 'hungry';
                else if (this.energy < 30) this.mood = 'tired';
                else if (this.health > 80 && this.hunger > 60) this.mood = 'happy';
            }
            
            explore() {
                this.targetX = Math.random() * 2800 + 100;
                this.targetY = Math.random() * 1800 + 100;
                this.mood = 'curious';
                this.lastAction = 'exploring';
                
                this.addThinkingLog('explore', '決定探索新區域', `我正在探索位置(${Math.floor(this.x)}, ${Math.floor(this.y)})周圍的區域`);
            }
            
            eat() {
                const nearbyResource = this.findNearbyResource('food');
                if (nearbyResource) {
                    this.targetX = nearbyResource.x;
                    this.targetY = nearbyResource.y;
                    
                    const dist = this.getDistance(nearbyResource);
                    if (dist < 30) {
                        // 第一次發現時命名
                        if (!nearbyResource.discovered) {
                            nearbyResource.discovered = true;
                            nearbyResource.name = this.nameResource(nearbyResource);
                            addEventLog(`🔍 ${this.name} 發現並命名了 "${nearbyResource.name}"`, 'discovery');
                            gameState.discoveries.push({
                                agent: this.name,
                                resource: nearbyResource.name,
                                symbol: nearbyResource.symbol,
                                time: gameState.worldTime
                            });
                        }
                        
                        // 檢查記憶中是否知道有毒
                        const isDangerous = this.memory.some(m => 
                            m.type === 'danger' && m.resource === nearbyResource.symbol
                        );
                        
                        if (isDangerous) {
                            this.speak(`${nearbyResource.name}有毒！我不要吃`);
                            this.mood = 'scared';
                            this.addThinkingLog('avoid-food', `發現${nearbyResource.name}`, '我知道這個有毒，決定避免');
                        } else {
                            nearbyResource.consume(this);
                            this.addThinkingLog('eat-food', `進食${nearbyResource.name}`, '食物很美味，營養豐富');
                        }
                    }
                } else {
                    this.explore(); // 找不到食物就探索
                    this.addThinkingLog('find-food', '尋找食物', '沒有找到附近的食物，決定擴大搜索範圍');
                }
                this.lastAction = 'eating';
            }
            
            drink() {
                const waterSource = this.findNearbyResource('water');
                if (waterSource) {
                    this.targetX = waterSource.x;
                    this.targetY = waterSource.y;
                    
                    const dist = this.getDistance(waterSource);
                    if (dist < 30) {
                        waterSource.consume(this);
                        this.speak('啊，好清涼的水！');
                        this.addThinkingLog('drink-water', '飲用水源', '補充了水分，感覺好多了');
                    }
                } else {
                    this.explore();
                    this.addThinkingLog('find-water', '尋找水源', '沒有找到附近的水源，決定探索更多區域');
                }
                this.lastAction = 'drinking';
            }
            
            startSleep() {
                this.isSleeping = true;
                this.lastSleepTime = gameState.worldTime;
                this.mood = 'tired';
                this.speak('Zzz...');
                this.addThinkingLog('sleep', '決定休息', '感到疲倦，決定睡一覺恢復體力');
                addEventLog(`😴 ${this.name} 開始睡覺`, 'survival');
            }
            
            sleep() {
                // 睡眠恢復能量
                this.energy = Math.min(100, this.energy + 2);
                
                // 睡滿一定時間後醒來 (2小時)
                if (this.energy >= 90 || (gameState.worldTime - this.lastSleepTime) > (2/60)) {
                    this.isSleeping = false;
                    this.mood = 'happy';
                    this.speak('早安！睡得真好');
                    this.addThinkingLog('wake-up', '醒來', '睡眠充足，現在精力充沛');
                    addEventLog(`⏰ ${this.name} 醒來了`, 'survival');
                }
            }
            
            rest() {
                this.health = Math.min(100, this.health + 2);
                this.energy = Math.min(100, this.energy + 1);
                this.mood = 'happy';
                this.lastAction = 'resting';
                
                this.addThinkingLog('rest', '休息恢復', '休息了一下，身體感覺好一些了');
            }
            
            socialize() {
                const nearbyAgent = this.findNearbyAgent();
                if (nearbyAgent) {
                    this.targetX = nearbyAgent.x;
                    this.targetY = nearbyAgent.y;
                    
                    const dist = this.getDistance(nearbyAgent);
                    if (dist < 50) {
                        // 建立關係
                        if (!this.relationships[nearbyAgent.id]) {
                            this.relationships[nearbyAgent.id] = 0;
                        }
                        this.relationships[nearbyAgent.id]++;
                        
                        // 分享知識
                        if (Math.random() < 0.3) {
                            this.shareKnowledge(nearbyAgent);
                        }
                        
                        this.mood = 'happy';
                        gameState.interactions++;
                        this.addThinkingLog('socialize', `與${nearbyAgent.name}交流`, '與其他AI交流很有趣');
                        addEventLog(`💬 ${this.name} 與 ${nearbyAgent.name} 互動`, 'social');
                    }
                }
                this.lastAction = 'socializing';
            }
            
            shareKnowledge(otherAgent) {
                // 分享關於危險資源的知識
                const dangerousResources = this.memory.filter(m => m.type === 'danger');
                if (dangerousResources.length > 0) {
                    const shared = dangerousResources[0];
                    otherAgent.memory.push(shared);
                    this.speak(`小心！${shared.name || shared.resource}有毒！`);
                    this.addThinkingLog('share-knowledge', `分享知識給${otherAgent.name}`, `提醒對方${shared.name}有毒`);
                    addEventLog(`📚 ${this.name} 向 ${otherAgent.name} 分享了知識`, 'social');
                }
            }
            
            die() {
                this.isAlive = false;
                this.mood = 'dead';
                gameState.deathCount++;
                this.addThinkingLog('death', '生命結束', `在第${gameState.currentDay}天因健康不足而死亡`);
                addEventLog(`💀 ${this.name} 死亡了 (健康:${Math.floor(this.health)} 飢餓:${Math.floor(this.hunger)} 口渴:${Math.floor(this.thirst)})`, 'survival');
                this.updateDisplay();
                updateStats();
            }
            
            findNearbyResource(type = null) {
                const visionRange = 150;
                let nearest = null;
                let minDist = Infinity;
                
                for (const resource of gameState.resources) {
                    // 過濾類型
                    if (type === 'food' && resource.type.nutrition === 0) continue;
                    if (type === 'water' && resource.type.hydration === 0) continue;
                    
                    const dist = this.getDistance(resource);
                    if (dist < visionRange && dist < minDist) {
                        minDist = dist;
                        nearest = resource;
                    }
                }
                
                return nearest;
            }
            
            findNearbyAgent() {
                let nearest = null;
                let minDist = Infinity;
                
                for (const agent of gameState.agents) {
                    if (agent.id === this.id || !agent.isAlive) continue;
                    
                    const dist = this.getDistance(agent);
                    if (dist < minDist && dist < 300) {
                        minDist = dist;
                        nearest = agent;
                    }
                }
                
                return nearest;
            }
            
            getDistance(target) {
                return Math.sqrt(
                    Math.pow(target.x - this.x, 2) + 
                    Math.pow(target.y - this.y, 2)
                );
            }
            
            nameResource(resource) {
                const prefixes = ['奇異', '神秘', '閃亮', '美味', '香甜', '野生', '罕見', '古老'];
                const suffixes = ['果', '草', '菌', '莓', '實', '花', '葉', '根'];
                
                const prefix = prefixes[Math.floor(Math.random() * prefixes.length)];
                const suffix = suffixes[Math.floor(Math.random() * suffixes.length)];
                
                return `${prefix}${suffix}`;
            }
            
            perceive() {
                const vision = {
                    terrain: this.getCurrentTerrain(),
                    nearbyAgents: [],
                    nearbyResources: [],
                    timeOfDay: gameState.dayNight,
                    hour: gameState.currentHour
                };
                
                // 感知其他 Agent
                for (const agent of gameState.agents) {
                    if (agent.id === this.id || !agent.isAlive) continue;
                    const dist = this.getDistance(agent);
                    if (dist < 200) {
                        vision.nearbyAgents.push({
                            name: agent.name,
                            mood: agent.mood,
                            health: agent.health,
                            distance: dist
                        });
                    }
                }
                
                // 感知資源
                for (const resource of gameState.resources) {
                    const dist = this.getDistance(resource);
                    if (dist < 150) {
                        vision.nearbyResources.push({
                            symbol: resource.symbol,
                            name: resource.name,
                            distance: dist
                        });
                    }
                }
                
                return vision;
            }
            
            getCurrentTerrain() {
                const x = Math.floor(this.x / 100);
                const y = Math.floor(this.y / 100);
                return gameState.terrainMap[y]?.[x] || TERRAIN_TYPES.GRASS;
            }
            
            wander() {
                const dx = (Math.random() - 0.5) * 100;
                const dy = (Math.random() - 0.5) * 100;
                this.targetX = Math.max(50, Math.min(2950, this.x + dx));
                this.targetY = Math.max(50, Math.min(1950, this.y + dy));
                this.lastAction = 'wandering';
                
                this.addThinkingLog('wander', '漫無目的地移動', '沒有特定目標，隨意探索');
            }
            
            speak(message) {
                const bubble = document.createElement('div');
                bubble.className = 'speech-bubble absolute bg-white text-black p-2 rounded-lg text-xs max-w-xs z-30';
                bubble.style.left = `${this.x + 20}px`;
                bubble.style.top = `${this.y - 30}px`;
                bubble.textContent = message;
                
                document.getElementById('worldMap').appendChild(bubble);
                
                setTimeout(() => bubble.remove(), 3000);
                
                addEventLog(`${this.name}: "${message}"`, 'speech');
            }
            
            move() {
                if (this.isSleeping || !this.isAlive) return;
                
                const speed = 5 * gameState.speed;
                const dx = this.targetX - this.x;
                const dy = this.targetY - this.y;
                const dist = Math.sqrt(dx * dx + dy * dy);
                
                if (dist > speed) {
                    this.x += (dx / dist) * speed;
                    this.y += (dy / dist) * speed;
                    
                    if (this.element) {
                        this.element.style.left = `${this.x}px`;
                        this.element.style.top = `${this.y}px`;
                    }
                }
            }
            
            updateDisplay() {
                if (!this.element) return;
                
                // 更新表情
                const emoji = this.element.querySelector('.text-2xl');
                if (emoji) {
                    emoji.textContent = this.getEmoji();
                    if (this.isSleeping) {
                        emoji.parentElement.classList.add('sleeping');
                        emoji.parentElement.classList.remove('floating-animation');
                    } else {
                        emoji.parentElement.classList.remove('sleeping');
                        emoji.parentElement.classList.add('floating-animation');
                    }
                }
                
                // 更新生命值條
                const healthBar = this.element.querySelector('.health-bar');
                if (healthBar) {
                    healthBar.style.width = `${this.health}%`;
                }
                
                // 更新狀態圖標
                const statusIcons = this.element.querySelector('.absolute.-top-2.-right-2');
                if (statusIcons) {
                    statusIcons.innerHTML = `
                        ${this.hunger < 30 ? '<span title="飢餓">🍽️</span>' : ''}
                        ${this.thirst < 30 ? '<span title="口渴">💧</span>' : ''}
                        ${this.energy < 30 ? '<span title="疲倦">😴</span>' : ''}
                    `;
                }
                
                // 死亡時降低透明度
                if (!this.isAlive) {
                    this.element.style.opacity = '0.5';
                }
            }
            
            showDetails() {

Langfuse 追踪: ${this.langfuseTraces?.length || 0} 條
                `;

            }

        function addEventLog(message, type = 'info') {
            const log = document.getElementById('eventLog');
            const entry = document.createElement('div');
            entry.className = `p-2 rounded log-entry ${type} ${
                type === 'system' ? 'bg-blue-900' :
                type === 'social' ? 'bg-purple-900' :
                type === 'survival' ? 'bg-green-900' :
                type === 'discovery' ? 'bg-yellow-900' :
                type === 'thinking' ? 'bg-orange-900' :
                type === 'speech' ? 'bg-gray-700' :
                'bg-gray-800'
            }`;
            
            const time = `${String(gameState.currentHour).padStart(2, '0')}:00`;
            entry.innerHTML = `
                <div class="text-xs text-gray-400">Day ${gameState.currentDay} ${time}</div>
                <div>${message}</div>
            `;
            
            log.insertBefore(entry, log.firstChild);
            
            // 保持日誌數量
            while (log.children.length > 100) {
                log.removeChild(log.lastChild);
            }
        }

        // 更新時間顯示
        function updateTimeDisplay() {
            // 計算當前時間 (1分鐘 = 2小時)
            const totalHours = Math.floor(gameState.worldTime * 2) % 24;
            gameState.currentHour = totalHours;
            gameState.currentDay = Math.floor(gameState.worldTime * 2 / 24) + 1;
            
            // 判斷日夜
            const isNight = gameState.currentHour >= 20 || gameState.currentHour < 6;
            const newDayNight = isNight ? 'night' : 'day';
            
            if (newDayNight !== gameState.dayNight) {
                gameState.dayNight = newDayNight;
                const container = document.getElementById('worldContainer');
                if (isNight) {
                    container.classList.remove('day-sky');
                    container.classList.add('night-sky');
                    addEventLog('🌙 夜晚降臨', 'system');
                } else {
                    container.classList.remove('night-sky');
                    container.classList.add('day-sky');
                    addEventLog('☀️ 太陽升起', 'system');
                }
                
                // 調整地形透明度
                document.querySelectorAll('[data-terrain]').forEach(cell => {
                    cell.style.opacity = isNight ? '0.2' : '0.4';
                });
            }
            
            // 更新顯示
            document.getElementById('dayIcon').textContent = isNight ? '🌙' : '☀️';
            document.getElementById('timeDisplay').textContent = 
                `Day ${gameState.currentDay} - ${String(gameState.currentHour).padStart(2, '0')}:00`;
        }

        // 移除 Agent
        function removeAgent(id) {
            const index = gameState.agents.findIndex(a => a.id == id);
            if (index !== -1) {
                const agent = gameState.agents[index];
                if (agent.element) {
                    agent.element.remove();
                }
                gameState.agents.splice(index, 1);
                updateAgentsList();
                updateStats();
                addEventLog(`${agent.name} 離開了世界`, 'system');
            }
        }

        // 顯示思考日誌內容
        function displayThinkingLog() {
            const agentId = document.getElementById('thinkingAgentSelect').value;
            const filter = document.getElementById('thinkingFilter').value;
            const logContainer = document.getElementById('thinkingLog');
            logContainer.innerHTML = '';
            
            let logs = [];
            
            if (agentId === 'all') {
                // 顯示所有 agent 的日誌
                gameState.agents.forEach(agent => {
                    agent.thinkingLog.forEach(log => {
                        logs.push({ ...log, agentName: agent.name });
                    });
                });
            } else {
                // 顯示特定 agent 的日誌
                const agent = gameState.agents.find(a => a.id == agentId);
                if (agent) {
                    logs = agent.thinkingLog.map(log => ({ ...log, agentName: agent.name }));
                }
            }
            
            // 過濾
            if (filter !== 'all') {
                logs = logs.filter(log => log.action === filter);
            }
            
            // 按時間排序
            logs.sort((a, b) => b.timestamp - a.timestamp);
            
            // 顯示日誌
            logs.forEach(log => {
                const entry = document.createElement('div');
                entry.className = 'p-2 bg-gray-700 rounded text-xs';
                entry.innerHTML = `
                    <div class="text-gray-400">Day ${log.day} ${String(log.hour).padStart(2, '0')}:00 - ${log.agentName}</div>
                    <div class="text-blue-400">行動: ${log.action}</div>
                    <div class="text-green-400">思考: ${log.response}</div>
                    <div class="text-yellow-400">成本: $${log.cost?.toFixed(6) || '0.000000'}</div>
                    <div class="text-gray-300">狀態: ❤️${log.gameState.health} 🍽️${log.gameState.hunger} 💧${log.gameState.thirst} ⚡${log.gameState.energy}</div>
                `;
                logContainer.appendChild(entry);
            });
        }

        // 遊戲循環
        function gameLoop() {
            if (!gameState.isPaused) {
                // 更新世界時間 (1分鐘 = 2小時)
                gameState.worldTime += (0.0167 * gameState.speed) / 30; // 調整時間比例
                updateTimeDisplay();
                
                // 更新所有 agents
                gameState.agents.forEach(agent => {
                    agent.move();
                    if (Math.random() < 0.1 * gameState.speed) { // 根據速度調整更新頻率
                        agent.update();
                    }
                });
                
                // 定期更新 UI
                if (Math.random() < 0.05) {
                    updateAgentsList();
                    updateStats();
                }
            }
            
            requestAnimationFrame(gameLoop);
        }
